----Obras en proceso

select *, to_char( "FechaCrea", 'DD/MM/YYYY') as fecha from "TblCpt"
where "UsrAlt"=26 and "CptTpo"=1 and "CptDsc" is not null and  "CptEstatus" != 1
and "FechaCrea" is not null and date_part('year', "FechaCrea")=2022
order by "EncIdn" asc

select a.*, to_char( a."FechaCrea", 'DD/MM/YYYY') as fecha from "TblCpt" as a
join "CptAux" as b on a."EncIdn"=b."EncIdn_id" 
where a."UsrAlt"=25 and a."CptTpo"=1 and a."CptDsc" is not null and  a."CptEstatus" != 1
and a."FechaCrea" is not null and date_part('year', a."FechaCrea")=2022
and (b."DatP101"+b."DatP102"+b."DatP103"+b."DatP104"+b."DatP105"+b."DatP106"+b."DatP107"+b."DatP108"+b."DatP109"+b."DatP110"+b."DatP111"+b."DatP112")>0
order by a."EncIdn" asc



----Obras terminadas    	verficar en cptaux del 101 al 112, que sea distinto de cero

select a.*, to_char( a."FechaCrea", 'DD/MM/YYYY') as fechacrea, to_char( a."FecMod", 'DD/MM/YYYY') as fechamod from "TblCpt" as a
join "CptAux" as b on a."EncIdn"=b."EncIdn_id" 
where a."UsrAlt"=25 and a."CptTpo"=2 and a."CptDsc" is not null and  a."CptEstatus" != 1
and a."FechaCrea" is not null and (date_part('year', a."FechaCrea")=2022 OR date_part('year', a."FecMod")=2022 )
and (b."DatP101"+b."DatP102"+b."DatP103"+b."DatP104"+b."DatP105"+b."DatP106"+b."DatP107"+b."DatP108"+b."DatP109"+b."DatP110"+b."DatP111"+b."DatP112")>0
order by a."EncIdn" asc

SELECT c."OrgDsc", c."OrgIdn", a."EncIdn_id", a."PDEIdn_id", d."PDECod" FROM
	(SELECT "EncIdn_id", "PDEIdn_id" 
	FROM "CptPDE" WHERE "EncIdn_id" IN(SELECT "EncIdn" FROM "TblCpt" 
	WHERE "UsrAlt"=25 AND "CptEstatus"!=1 ) AND "PDEIdn_id"!=0 
	GROUP BY "EncIdn_id","PDEIdn_id" ORDER BY "EncIdn_id" ASC ) as a
JOIN "OrgPDE" as b ON a."PDEIdn_id"=b."PDEIdn_id" 
JOIN "TblOrg" as c ON b."OrgIdn_id"=c."OrgIdn"
JOIN "TblPDE" as d ON a."PDEIdn_id"=d."PDEIdn"
WHERE b."OrgIdn_id"=25
ORDER BY a."EncIdn_id"
